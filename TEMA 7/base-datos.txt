CREATE DATABASE IF NOT EXISTS UniversidadDB;

-- Crear la tabla Estudiantes
CREATE TABLE IF NOT EXISTS Estudiantes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL
) ENGINE=InnoDB;

-- Crear la tabla Cursos
CREATE TABLE IF NOT EXISTS Cursos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    creditos INT NOT NULL
) ENGINE=InnoDB;

-- Crear la tabla Matriculas con claves foráneas a Estudiantes y Cursos
CREATE TABLE IF NOT EXISTS Matriculas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    estudiante_id INT NOT NULL,
    curso_id INT NOT NULL,
    fecha_matricula DATE NOT NULL,
    CONSTRAINT fk_estudiante
      FOREIGN KEY (estudiante_id) REFERENCES Estudiantes(id)
      ON DELETE CASCADE
      ON UPDATE CASCADE,
    CONSTRAINT fk_curso
      FOREIGN KEY (curso_id) REFERENCES Cursos(id)
      ON DELETE CASCADE
      ON UPDATE CASCADE
) ENGINE=InnoDB;

-- Crear el procedimiento almacenado sp_updateMatricula
DELIMITER //
CREATE PROCEDURE sp_updateMatricula(
    IN p_idMatricula INT,
    IN p_fechaMatricula DATE
)
BEGIN
    UPDATE Matriculas
    SET fecha_matricula = p_fechaMatricula
    WHERE id = p_idMatricula;
END //
DELIMITER ;
